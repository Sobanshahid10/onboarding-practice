    <script>

        function fillInput(text) { document.getElementById('question').value = text; }


        async function testBot() {

            const questionInput = document.getElementById('question');

            const resDiv = document.getElementById('response');

            const question = questionInput.value;


            if (!question) return alert("Please enter a question");


            resDiv.innerHTML = '<span class="loading">Bot is thinking...</span>';

            

            try {

                const response = await fetch('http://localhost:5678/webhook-test/customer-support', {

                    method: 'POST',

                    headers: { 'Content-Type': 'application/json' },

                    body: JSON.stringify({ question: question })

                });


                const data = await response.json();

                resDiv.innerText = data.answer || "Success! Check your Google Sheet.";

            } catch (error) {

                resDiv.innerText = "Error: Could not reach the webhook. Ensure n8n is running.";

                console.error(error);

            }

        }

    </script>
